/* ==========================================================
SITE HEADER
This is the navigation bar that appears at the top of the
About page and eventually the Updates page as well.
========================================================== */

.site-header {

    /* ------------------------------------------------------
    Keep the header visible while scrolling.

    "sticky" means the header behaves normally until the
    visitor scrolls, then it stays at the top of the screen.
    ------------------------------------------------------ */
    position: sticky;

    /* Stick to the very top of the browser window */
    top: 0;

    /* Make sure the header stays above the page content */
    z-index: 1000;

    /* White background so text remains easy to read */
    background: #ffffff;

    /* Bottom border gives the header some separation */
    border-bottom: 1px solid #d2d8e0;

    /* Small shadow underneath the header */
    box-shadow: 0 2px 10px rgba(31, 41, 55, 0.08);

    /* Internal spacing */
    padding: 12px 10px;

}
